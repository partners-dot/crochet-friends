const products = document.querySelectorAll('.product'); let selectedProduct = 'potato'; message.addEventListener('input', () => charCount.textContent = message.value.length); products.forEach(p => p.addEventListener('click', () => { products.forEach(x => x.classList.remove('selected')); p.classList.add('selected'); selectedProduct = p.dataset.product; })); form.addEventListener('submit', async (e) => { e.preventDefault(); const text = message.value.trim(); if (!text) return; document.getElementById('form').classList.add('hidden'); document.getElementById('loading').classList.remove('hidden'); progressText.textContent = 'Starting video generation...'; try { const startRes = await fetch('/api/generate-video', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: text, character: selectedProduct }) }); const startData = await startRes.json(); if (!startRes.ok) throw new Error(startData.error || 'Failed to start generation'); const operationId = startData.operationId; console.log('Generation started, operationId:', operationId); progressText.textContent = 'Video is being created...'; let videoUrl = null; let attempts = 0; let consecutiveFailures = 0; const maxAttempts = 60; while (!videoUrl && attempts < maxAttempts) { await new Promise(r => setTimeout(r, 5000)); attempts++; progressText.textContent = 'Creating magic... (' + (attempts * 5) + 's)'; const statusRes = await fetch('/api/check-status?operationId=' + encodeURIComponent(operationId)); const statusData = await statusRes.json(); if (!statusRes.ok) { console.error('Status check failed:', statusData); consecutiveFailures++; if (consecutiveFailures >= 3) { throw new Error(statusData.details || statusData.error || 'Video generation failed - server error'); } continue; } if (statusData.status === 'done') { consecutiveFailures = 0; videoUrl = statusData.videoUrl; } else if (statusData.status === 'error') { throw new Error(statusData.error || 'Video generation failed'); } } if (!videoUrl) { throw new Error('Video generation timed out after 5 minutes'); } document.getElementById('video').src = videoUrl; document.getElementById('downloadBtn').href = videoUrl; document.getElementById('downloadBtn').download = 'crochet-' + selectedProduct + '-message.mp4'; document.getElementById('loading').classList.add('hidden'); document.getElementById('result').classList.remove('hidden'); } catch (err) { console.error(err); document.getElementById('errorText').textContent = err.message || 'Something went wrong. Please try again.'; document.getElementById('loading').classList.add('hidden'); document.getElementById('error').classList.remove('hidden'); } }); function reset() { ['loading', 'result', 'error'].forEach(id => document.getElementById(id).classList.add('hidden')); document.getElementById('form').classList.remove('hidden'); }
